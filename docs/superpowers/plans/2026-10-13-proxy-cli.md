# Smruti v1 Proxy and CLI Implementation Plan (Plan 2 of 3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the memory core usable with zero code changes. `smruti proxy` sits between any OpenAI-compatible client (OpenCode, Open WebUI, Continue, scripts) and any local runtime (Ollama, llama.cpp, LM Studio, vLLM). It injects recalled memory, records new turns (including tool results), and consolidates in the background. A `smruti` CLI inspects and manages memory, and the docs are rewritten for the local-LLM product.

**Scope decision:** no MCP server and no agent hooks in v1. Clients that support a custom OpenAI base URL get memory through the proxy. Tool calls and results already flow through the conversation the proxy relays, so they are captured there. A thin optional MCP server can come later on demand.

**Architecture:** `smruti/serve/capture.py` (pure: which turns are new, SSE text accumulation) → `smruti/serve/scheduler.py` (background sleep thread) → `smruti/serve/proxy.py` (Starlette app: chat completions with recall plus recording, a passthrough for every other `/v1/*` route, fail-open) → `smruti/cli.py` (Typer). It uses only the public `Memory` API from Plan 1, plus two small additions (`Memory.stats`, `Memory.due_spaces`).

**Tech Stack:** Starlette, uvicorn, httpx (AsyncClient), Typer, pytest with `starlette.testclient.TestClient` and `httpx.MockTransport` as the fake upstream.

**Running tests:** `python -m pytest -q` in an environment with `pip install -e ".[dev]"`.

```python
"""
capture.py - What to record from proxied traffic. Clients resend the whole conversation on
every request, so only the turns after the last assistant message are new: the user's latest
message and any tool results (paired with the call that produced them). Replies are read from
JSON responses or accumulated from SSE streams as they pass through.
"""

import codecs
import json

from smruti.messages import message_text


def _call_label(call: dict) -> str:
    function = call.get("function") or {}
    return f"{function.get('name', 'tool')}({function.get('arguments', '')})"


def new_turns(messages: list[dict]) -> list[tuple[str, str]]:
    """(role, text) for the new user and tool turns of a request."""
    last_assistant = max((i for i, m in enumerate(messages) if m.get("role") == "assistant"), default=-1)
    calls = {}
    if last_assistant >= 0:
        calls = {c.get("id"): _call_label(c) for c in messages[last_assistant].get("tool_calls") or []}
    turns = []
    for message in messages[last_assistant + 1:]:
        role, text = message.get("role"), message_text(message).strip()
        if role == "user" and text:
            turns.append(("user", text))
        elif role == "tool":
            turns.append(("tool", f"{calls.get(message.get('tool_call_id'), 'tool call')} -> {text}"))
    return turns


def reply_text(body: bytes) -> str:
    """Assistant text from a non-streaming chat completion body ('' if absent or unparsable)."""
    try:
        return message_text(json.loads(body)["choices"][0]["message"]).strip()
    except (ValueError, KeyError, IndexError, TypeError):
        return ""


class StreamAccumulator:
    """Collects assistant text from OpenAI-style server-sent event chunks."""

    def __init__(self):
        self._decoder = codecs.getincrementaldecoder("utf-8")(errors="replace")
        self._pending = ""
        self._parts: list[str] = []

    def feed(self, chunk: bytes) -> None:
        self._pending += self._decoder.decode(chunk)
        *lines, self._pending = self._pending.split("\n")
        for line in lines:
            line = line.strip()
            if not line.startswith("data:") or line == "data: [DONE]":
                continue
            try:
                delta = json.loads(line[5:])["choices"][0].get("delta") or {}
            except (ValueError, KeyError, IndexError, TypeError, AttributeError):
                continue
            if isinstance(delta.get("content"), str):
                self._parts.append(delta["content"])

    @property
    def text(self) -> str:
        return "".join(self._parts).strip()
```

```python
"""
scheduler.py - Background sleep. While the proxy serves requests, a daemon thread periodically
consolidates every space that is due (enough pending turns, or idle). Failures are logged and
retried on the next poll; they never reach a request.
"""

import logging
import threading

logger = logging.getLogger(__name__)


class SleepScheduler:
    def __init__(self, memory, poll_seconds: float):
        self.memory = memory
        self.poll_seconds = poll_seconds
        self._stop = threading.Event()
        self._thread: threading.Thread | None = None

    @property
    def running(self) -> bool:
        return self._thread is not None and self._thread.is_alive()

    def run_once(self, now=None) -> list[str]:
        """Sleeps every due space; returns the spaces consolidated."""
        if self.memory.llm is None or self.memory.model is None:
            return []  # the model is learned from the first proxied request
        done = []
        for space in self.memory.due_spaces(now):
            try:
                self.memory.sleep(space, now=now)
                done.append(space)
            except Exception:
                logger.exception("sleep failed for space %r; will retry", space)
        return done

    def start(self) -> None:
        self._stop.clear()
        self._thread = threading.Thread(target=self._loop, name="smruti-sleep", daemon=True)
        self._thread.start()

    def stop(self) -> None:
        self._stop.set()
        if self._thread is not None:
            self._thread.join(timeout=5)

    def _loop(self) -> None:
        while not self._stop.wait(self.poll_seconds):
            self.run_once()
```

```python
"""
proxy.py - OpenAI-compatible memory proxy. Sits between any chat client and any
OpenAI-compatible runtime: recalled memory is merged into each chat request, new user and
tool turns plus the reply are recorded, and every other /v1 endpoint passes through untouched.
Memory failures never break a request: recall errors forward the original request, recording
errors are logged.
"""

import logging
from contextlib import asynccontextmanager

import httpx
from starlette.applications import Starlette
from starlette.background import BackgroundTask
from starlette.concurrency import run_in_threadpool
from starlette.requests import Request
from starlette.responses import JSONResponse, Response, StreamingResponse
from starlette.routing import Route

from smruti.memory import Memory
from smruti.serve.capture import StreamAccumulator, new_turns, reply_text
from smruti.serve.scheduler import SleepScheduler

logger = logging.getLogger(__name__)

SPACE_HEADER = "x-smruti-space"
_DROPPED_HEADERS = {"host", "content-length", "connection", "transfer-encoding", "accept-encoding", SPACE_HEADER}


def create_app(memory: Memory, upstream: str, default_space: str = "default",
               sleep_poll_seconds: float | None = 30.0, transport: httpx.AsyncBaseTransport | None = None) -> Starlette:
    client = httpx.AsyncClient(base_url=upstream.rstrip("/"), timeout=None, transport=transport)
    scheduler = SleepScheduler(memory, sleep_poll_seconds) if sleep_poll_seconds else None

    def forwarded_headers(request: Request) -> dict[str, str]:
        return {k: v for k, v in request.headers.items() if k.lower() not in _DROPPED_HEADERS}

    def record(space: str, turns: list[tuple[str, str]], reply: str) -> None:
        try:
            for role, text in turns:
                memory.observe(role, text, space)
            if reply:
                memory.observe("assistant", reply, space)
        except Exception:
            logger.exception("recording turns failed for space %r", space)

    def upstream_error(exc: Exception) -> JSONResponse:
        return JSONResponse({"error": {"message": f"upstream unreachable: {exc}", "type": "upstream_error"}},
                            status_code=502)

    async def chat_completions(request: Request) -> Response:
        body = await request.json()
        messages = body.get("messages") or []
        space = request.headers.get(SPACE_HEADER) or body.get("user") or default_space
        if memory.model is None and body.get("model"):
            memory.model = body["model"]  # sleep consolidates with the model clients use
        try:
            body = {**body, "messages": await run_in_threadpool(memory.build_context, messages, space)}
        except Exception:
            logger.exception("recall failed; forwarding the request unchanged")
        turns = new_turns(messages)

        outgoing = client.build_request("POST", "/chat/completions", json=body, headers=forwarded_headers(request))
        try:
            response = await client.send(outgoing, stream=True)
        except httpx.HTTPError as exc:
            return upstream_error(exc)

        content_type = response.headers.get("content-type")
        if response.status_code >= 400 or not body.get("stream"):
            content = await response.aread()
            await response.aclose()
            task = BackgroundTask(record, space, turns, reply_text(content)) if response.status_code < 400 else None
            return Response(content, status_code=response.status_code, media_type=content_type, background=task)

        accumulator = StreamAccumulator()

        async def relay():
            try:
                async for chunk in response.aiter_bytes():
                    accumulator.feed(chunk)
                    yield chunk
            finally:
                await response.aclose()

        return StreamingResponse(relay(), status_code=response.status_code,
                                 media_type=content_type or "text/event-stream",
                                 background=BackgroundTask(lambda: record(space, turns, accumulator.text)))

    async def passthrough(request: Request) -> Response:
        outgoing = client.build_request(request.method, "/" + request.path_params["path"],
                                        content=await request.body(), headers=forwarded_headers(request),
                                        params=request.query_params)
        try:
            response = await client.send(outgoing)
        except httpx.HTTPError as exc:
            return upstream_error(exc)
        return Response(response.content, status_code=response.status_code,
                        media_type=response.headers.get("content-type"))

    @asynccontextmanager
    async def lifespan(app):
        if scheduler is not None:
            scheduler.start()
        try:
            yield
        finally:
            if scheduler is not None:
                scheduler.stop()
            await client.aclose()

    return Starlette(
        routes=[
            Route("/v1/chat/completions", chat_completions, methods=["POST"]),
            Route("/v1/{path:path}", passthrough, methods=["GET", "POST", "PUT", "PATCH", "DELETE"]),
        ],
        lifespan=lifespan,
    )
```

```python
"""cli.py - The `smruti` command: run the memory proxy, consolidate, and inspect memory."""

import json
import logging

import typer

from smruti.memory import Memory

DEFAULT_UPSTREAM = "http://localhost:11434/v1"  # Ollama's OpenAI-compatible endpoint

app = typer.Typer(help="smruti: local neuro-symbolic long-term memory for small LLMs.",
                  no_args_is_help=True, add_completion=False)


def _memory(upstream: str | None = None, model: str | None = None) -> Memory:
    return Memory(upstream=upstream, model=model)


@app.command()
def proxy(
    upstream: str = typer.Option(DEFAULT_UPSTREAM, help="OpenAI-compatible base URL of your local runtime."),
    host: str = typer.Option("127.0.0.1", help="Interface to listen on."),
    port: int = typer.Option(8080, help="Port to listen on."),
    model: str | None = typer.Option(None, help="Model for sleep consolidation (default: the model clients request)."),
    space: str = typer.Option("default", help="Memory space for requests that name none."),
    sleep_poll: float = typer.Option(30.0, help="Seconds between background sleep checks; 0 disables."),
):
    """Run the memory proxy; point any OpenAI-compatible client at http://HOST:PORT/v1."""
    import uvicorn

    from smruti.serve.proxy import create_app

    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(name)s %(levelname)s %(message)s")
    memory = _memory(upstream, model)
    typer.echo(f"smruti proxy on http://{host}:{port}/v1 -> {upstream} (memory: {memory.config.db_path})")
    app_ = create_app(memory, upstream, default_space=space, sleep_poll_seconds=sleep_poll or None)
    try:
        uvicorn.run(app_, host=host, port=port, log_level="warning")
    finally:
        memory.close()


@app.command()
def sleep(
    model: str = typer.Option(..., help="Model to consolidate with, e.g. llama3.2:3b."),
    upstream: str = typer.Option(DEFAULT_UPSTREAM, help="OpenAI-compatible base URL of your local runtime."),
    space: str = typer.Option("default", help="Memory space."),
):
    """Consolidate pending turns into long-term memory now."""
    memory = _memory(upstream, model)
    try:
        report = memory.sleep(space)
    finally:
        memory.close()
    typer.echo(f"Consolidated {report.episodes} turns in {report.batches} batches "
               f"({report.failed_batches} failed): {report.consolidation}")


@app.command()
def facts(
    entity: str | None = typer.Argument(None, help="Only facts about this entity, e.g. user."),
    space: str = typer.Option("default", help="Memory space."),
    history: bool = typer.Option(False, "--history", help="Include facts that are no longer true."),
):
    """List remembered facts."""
    memory = _memory()
    try:
        for fact in memory.facts(entity, space, include_closed=history):
            note = "" if fact.valid_to is None else "  (no longer true)"
            typer.echo(f"[{fact.id}] {fact.text}  confidence={fact.confidence:.2f}{note}")
    finally:
        memory.close()


@app.command()
def recall(
    query: str = typer.Argument(..., help="A message, as a user would send it."),
    space: str = typer.Option("default", help="Memory space."),
    budget: int | None = typer.Option(None, help="Token budget (default from config)."),
):
    """Show the memory block that would be added to the prompt for QUERY."""
    memory = _memory()
    try:
        context = memory.build_context([{"role": "user", "content": query}], space, budget)
    finally:
        memory.close()
    typer.echo(context[0]["content"] if context[0]["role"] == "system" else "(nothing recalled)")


@app.command()
def forget(
    fact_id: int = typer.Argument(..., help="Fact id, as shown by `smruti facts`."),
    space: str = typer.Option("default", help="Memory space."),
):
    """Mark a fact as no longer true."""
    memory = _memory()
    try:
        forgotten = memory.forget(fact_id, space)
    finally:
        memory.close()
    typer.echo("Forgotten." if forgotten else "No current fact with that id in this space.")
    raise typer.Exit(0 if forgotten else 1)


@app.command()
def stats(space: str = typer.Option("default", help="Memory space.")):
    """Counts of what is stored, as JSON."""
    memory = _memory()
    try:
        data = {"db": str(memory.config.db_path), "space": space, **memory.stats(space)}
    finally:
        memory.close()
    typer.echo(json.dumps(data, indent=2))
```
